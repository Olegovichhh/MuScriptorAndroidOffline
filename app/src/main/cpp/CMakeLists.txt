cmake_minimum_required(VERSION 3.24)
project(muscriptor_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
include(FetchContent)
FetchContent_Declare(muscriptor GIT_REPOSITORY https://github.com/DamRsn/muscriptor.cpp.git GIT_TAG main SOURCE_SUBDIR cpp)
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_VULKAN OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_METAL OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_NATIVE OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(muscriptor)
add_library(muscriptor_android SHARED native-lib.cpp)
target_link_libraries(muscriptor_android PRIVATE muscriptor_ggml android log)
